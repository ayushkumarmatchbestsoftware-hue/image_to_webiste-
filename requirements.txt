# Everything this service imports, and nothing else.
#
# The previous list carried a different application: Postgres and SQLAlchemy,
# Mongo, Redis, boto3, JWT and bcrypt, nine OpenTelemetry packages and a second
# web server. None of it was imported by a single line of this codebase. That
# is not only clutter - it is a slower build, a larger image, and a much larger
# surface to keep patched for code that does not exist.
#
# Checked by parsing every import in the tree rather than by reading, so what
# is here is what the code actually asks for.

# ── Web layer ────────────────────────────────────────────────────────────────
fastapi==0.136.3
uvicorn==0.49.0
# Not imported directly. FastAPI needs it to parse the multipart upload the
# seller's photograph arrives in, and fails at request time without it.
python-multipart==0.0.32

# ── Rendering ────────────────────────────────────────────────────────────────
jinja2==3.1.6
# Used directly by core/composition.py for its escaping primitives.
markupsafe==3.0.3

# ── Photographs ──────────────────────────────────────────────────────────────
pillow==12.3.0
# HEIC is what phones shoot by default, and Pillow cannot open it alone.
# core/__init__.py registers this on import.
pillow-heif==1.5.0

# ── Model access ─────────────────────────────────────────────────────────────
# Every supported provider speaks the OpenAI wire format, so this is one client
# pointed at a different base_url (see core/llm.py).
openai==2.46.0

# ── Image treatment ──────────────────────────────────────────────────────────
# numpy backs the background measurement and the connected-component despeckle
# in core/imagedirector.py.
#
# rembg and onnxruntime were here and are deliberately gone. They existed for
# one function - a local background cutout - and charged the whole service for
# it: onnxruntime, scipy, scikit-image, pymatting, pooch, jsonschema and tqdm
# came with them, and an inference session sat in memory for the life of the
# process whether a seller uploaded anything or not. Background removal is done
# by the image model the key already pays for, and the local fallback is a PIL
# flood fill that allocates nothing until it is called.
numpy==2.5.1

# ── Configuration ────────────────────────────────────────────────────────────
python-dotenv==1.2.2
